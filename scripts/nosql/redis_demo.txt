ECHO '=== NHÓM 1. OTP QUÊN MẬT KHẨU: hash + PEXPIRE (RedisPasswordResetOtpStore) ==='

ECHO '-- Q-17 [CƠ BẢN] Lưu OTP: HSET (code, expiresAt tính bằng mili-giây) + PEXPIRE 300000 (5 phút) + PTTL --'
ECHO 'Access pattern: người dùng bấm Quên mật khẩu, hệ thống lưu mã 6 số; Redis tự xóa key sau 5 phút nên không cần job dọn dẹp.'
DEL "ledgermind:password-reset-otp:sv01@huit.edu.vn"
HSET "ledgermind:password-reset-otp:sv01@huit.edu.vn" code 482931 expiresAt 4102444800000
PEXPIRE "ledgermind:password-reset-otp:sv01@huit.edu.vn" 300000
PTTL "ledgermind:password-reset-otp:sv01@huit.edu.vn"
HGETALL "ledgermind:password-reset-otp:sv01@huit.edu.vn"

ECHO '-- Q-18 [CƠ BẢN] Nhập sai mã: HINCRBY failedAttempts, đọc 1 trường bằng HGET; hết hạn thì key biến mất --'
ECHO 'Access pattern: đếm số lần đoán sai trên cùng key (app xóa OTP khi sai quá số lần cho phép). PEXPIREAT về quá khứ = hết hạn ngay.'
HINCRBY "ledgermind:password-reset-otp:sv01@huit.edu.vn" failedAttempts 1
HGET "ledgermind:password-reset-otp:sv01@huit.edu.vn" failedAttempts
EXISTS "ledgermind:password-reset-otp:sv01@huit.edu.vn"
PEXPIREAT "ledgermind:password-reset-otp:sv01@huit.edu.vn" 1
EXISTS "ledgermind:password-reset-otp:sv01@huit.edu.vn"
HGETALL "ledgermind:password-reset-otp:sv01@huit.edu.vn"

ECHO '=== NHÓM 2. CACHE THỐNG KÊ: String JSON + EX (CacheConfig, khóa <prefix>:v1:<cache>:<key>) ==='

ECHO '-- Q-19 [CƠ BẢN] Cache dashboard của user 1: SET ... EX, GET, TTL --'
ECHO 'Access pattern: lần đầu đọc PostgreSQL rồi ghi cache; lần sau trả thẳng từ Redis. Giá trị là JSON rút gọn (ví dụ minh họa).'
SET "ledgermind:default:v1:user-dashboard:1" '{"totalBalance":12500000,"monthlyExpense":3150000,"monthlyIncome":15000000}' EX 285
GET "ledgermind:default:v1:user-dashboard:1"
TTL "ledgermind:default:v1:user-dashboard:1"
TYPE "ledgermind:default:v1:user-dashboard:1"

ECHO '-- Q-20 [CƠ BẢN] TTL jitter: các entry có TTL khác nhau để không hết hạn cùng lúc (chống cache stampede); xóa key = evict --'
ECHO 'Access pattern: CacheConfig.JitteredTtl rút ngắn TTL thêm tối đa vài phần trăm một cách ngẫu nhiên. Ở đây giả lập bằng các giá trị EX khác nhau.'
SET "ledgermind:default:v1:user-dashboard:2" '{"totalBalance":4200000}' EX 270
SET "ledgermind:default:v1:user-dashboard:3" '{"totalBalance":9800000}' EX 298
SET "ledgermind:default:v1:user-popular-vendors:1" '[{"vendorName":"Highlands Coffee","visits":12}]' EX 600
TTL "ledgermind:default:v1:user-dashboard:2"
TTL "ledgermind:default:v1:user-dashboard:3"
DEL "ledgermind:default:v1:user-dashboard:3"
EXISTS "ledgermind:default:v1:user-dashboard:3"

ECHO '-- Q-21 [CƠ BẢN] SCAN MATCH thay cho KEYS: liệt kê key của một cache, không chặn Redis --'
ECHO 'Access pattern: xóa cả một cache admin (CacheConfig, BatchStrategies.scan). KEYS bị đổi tên thành rỗng trên stack thật. Trả về: con trỏ mới + danh sách; con trỏ 0 = hết.'

ECHO '-- Q-24 [NÂNG CAO] Thứ hạng của một user: ZREVRANK (0 = đầu bảng), ZSCORE, ZCOUNT trong khoảng số tiền --'
ECHO 'Access pattern: user 1 đứng thứ mấy trong tháng; có bao nhiêu user chi từ 3 triệu đến 7 triệu.'
ZREVRANK "ledgermind:leaderboard:spend:2026-10" 1

ECHO '-- Q-25 [NÂNG CAO] ZINCRBY (cộng dồn) khác ZADD (ghi tuyệt đối): app dùng ZADD để tự lành khi mất/lặp sự kiện --'
ECHO 'Access pattern: user 5 thêm giao dịch 500000. ZINCRBY cho 3800000 nhưng nếu sự kiện bị lặp thì cộng hai lần; app đọc lại tổng từ PostgreSQL rồi ZADD giá trị tuyệt đối.'

ECHO 'Access pattern: dựng bảng xếp hạng tháng 9 và gắn TTL cùng lúc; EXEC trả về danh sách kết quả từng lệnh (QUEUED là lệnh đang xếp hàng). DISCARD để hủy.'
