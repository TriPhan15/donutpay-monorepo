# ---------------------------------------------------------------------------
# Q-32 [CƠ BẢN] Giao dịch của MỘT người dùng, mới nhất trước (term trên userId + sort)
# Access pattern: mọi truy vấn của app đều nằm trong filter userId của người gọi (không bao giờ lộ dữ liệu người khác).
# ---------------------------------------------------------------------------
GET transactions/_search
{
  "size": 3,
  "query": { "bool": { "filter": [ { "term": { "userId": 1 } } ] } },
  "sort": [ { "transactionDate": "desc" } ],
  "_source": [ "vendorName", "note", "amount", "transactionDate" ]
}

# ---------------------------------------------------------------------------
# Q-33 [CƠ BẢN] match trên trường đã bỏ dấu: gõ "ca phe" ra được danh mục "Cà phê"
# Access pattern: người dùng gõ không dấu. Trường categoryName dùng analyzer vi_folded.
# ---------------------------------------------------------------------------
GET transactions/_search
{
  "size": 5,
  "query": { "bool": {
    "filter": [ { "term": { "userId": 2 } } ],
    "must": [ { "match": { "categoryName": "ca phe" } } ]
  } },
  "_source": [ "categoryName", "vendorName", "amount" ]
}

# ---------------------------------------------------------------------------

GET transactions/_analyze

  "analyzer": "vi_folded",
  "text": "Phở Bò Hòa Pasteur"

GET transactions/_analyze
{
  "analyzer": "vi_autocomplete",
  "text": "Highlands"
}

# ---------------------------------------------------------------------------
# Q-35 [NÂNG CAO] TRUY VẤN THẬT CỦA APP cho GET /api/transactions/search?q=pho
# (ElasticsearchTransactionSearchEngine.hits): bool { filter userId, should [ multi_match chính xác/tiền tố trên cả
# trường autocomplete (boost 2), multi_match mờ fuzziness AUTO:4,7 + prefix_length 1 (chữ đầu phải đúng) chỉ trên trường nguyên từ ], minimum_should_match 1 },
# sắp xếp theo _score rồi transactionDate giảm dần. Phải ra các dòng "Phở ..." của user 1 và KHÔNG ra user khác.
# Kết quả: chỉ các dòng "Phở ..."/"Phố ..." (và tiền tố "Ph...") của user 1; từ "pho" có 3 chữ nên nhánh MỜ không còn kéo
# "Chợ" hay "Phí" vào (AUTO:4,7 không cho sai với từ dưới 4 chữ).
# ---------------------------------------------------------------------------
GET transactions/_search
{
  "from": 0,
  "size": 20,
  "track_total_hits": true,
  "query": {
    "bool": {
      "filter": [ { "term": { "userId": 1 } } ],
      "should": [
        { "multi_match": {
            "query": "pho",
            "fields": [ "vendorName^3", "vendorName.autocomplete^2", "note^2", "note.autocomplete", "categoryName", "location", "paymentMethod" ],
            "operator": "and",
            "boost": 2.0
        } },
        { "multi_match": {
            "query": "pho",
            "fields": [ "vendorName^3", "note^2", "categoryName", "location", "paymentMethod" ],
            "fuzziness": "AUTO:4,7",
            "prefix_length": 1,
            "operator": "and"
        } }
      ],
      "minimum_should_match": "1"
    }
  },
  "sort": [ { "_score": "desc" }, { "transactionDate": "desc" } ],
  "_source": [ "vendorName", "note", "transactionDate" ]
}

# ---------------------------------------------------------------------------
# Q-36 [NÂNG CAO] Tìm mờ (fuzzy multi_match): gõ sai "highlans cofee" vẫn ra "Highlands Coffee"
# Access pattern: lỗi gõ phím. fuzziness AUTO:4,7 không cho sai với từ dưới 4 chữ, cho sai 1 ký tự với từ 4-7 chữ, 2 ký tự với từ từ 8 chữ trở lên.
# ---------------------------------------------------------------------------
GET transactions/_search
{
  "query": { "bool": {
    "filter": [ { "term": { "userId": 1 } } ],
    "must": [ { "multi_match": {
        "query": "highlans cofee",
        "fields": [ "vendorName^3", "note^2", "categoryName", "location", "paymentMethod" ],
        "fuzziness": "AUTO:4,7",
        "prefix_length": 1,
        "operator": "and"
    } } ]
  } },
  "_source": [ "vendorName", "note", "amount" ]
}

# ---------------------------------------------------------------------------
# Q-37 [NÂNG CAO] Gợi ý khi đang gõ (autocomplete): "hig" khớp tiền tố "Highlands" nhờ edge_ngram lúc LẬP chỉ mục
# Access pattern: ô tìm kiếm gợi ý từng chữ. Lúc lập index vi_autocomplete cắt "highlands" thành hi, hig, high, ...;
# lúc tìm dùng vi_folded (không cắt) nên "hig" chỉ khớp những từ BẮT ĐẦU bằng hig.
# ---------------------------------------------------------------------------
GET transactions/_search
{
  "size": 5,
  "query": { "bool": {
    "filter": [ { "term": { "userId": 1 } } ],
    "must": [ { "multi_match": {
        "query": "hig",
        "fields": [ "vendorName.autocomplete^2", "note.autocomplete" ]
    } } ]
  } },
  "_source": [ "vendorName", "note" ]
}

# ---------------------------------------------------------------------------
# Q-38 [NÂNG CAO] bool filter theo người dùng + khoảng ngày (range) + chỉ khoản chi, xếp theo số tiền giảm dần
# Access pattern: "chi tiêu của tôi trong tháng 9/2026". filter không tính điểm và được cache; range dùng gte/lt.
# ---------------------------------------------------------------------------
GET transactions/_search
{
  "size": 5,
  "query": { "bool": { "filter": [
    { "term": { "userId": 1 } },
    { "term": { "expense": true } },
    { "range": { "transactionDate": { "gte": "2026-09-01T00:00:00.000", "lt": "2026-10-01T00:00:00.000" } } }
  ] } },
  "sort": [ { "amount": "desc" } ],
  "_source": [ "vendorName", "categoryName", "amount", "transactionDate" ]
}
